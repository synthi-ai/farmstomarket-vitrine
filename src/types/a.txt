// GSAP-like 3D animations (simulated with CSS transforms)
const useGSAP3D = (ref: React.RefObject<HTMLElement | null>, options: any) => {
  useEffect(() => {
    if (!ref.current || !options.enabled) return;

    const element = ref.current;
    let animationId: number;

    const animate = () => {
      const time = Date.now() * 0.001;
      
      // 3D rotation and floating with customizable parameters
      const rotateX = Math.sin(time * (options.speedX || 0.5)) * (options.rangeX || 5);
      const rotateY = Math.cos(time * (options.speedY || 0.3)) * (options.rangeY || 8);
      const rotateZ = Math.sin(time * (options.speedZ || 0.2)) * (options.rangeZ || 3);
      const translateY = Math.sin(time * (options.floatSpeed || 0.8)) * (options.floatRange || 10);
      const scale = 1 + Math.sin(time * (options.scaleSpeed || 0.6)) * (options.scaleRange || 0.02);

      element.style.transform = `
        perspective(1000px) 
        rotateX(${rotateX}deg) 
        rotateY(${rotateY}deg) 
        rotateZ(${rotateZ}deg)
        translateY(${translateY}px) 
        scale(${scale})
      `;

      animationId = requestAnimationFrame(animate);
    };

    animate();

    return () => {
      if (animationId) {
        cancelAnimationFrame(animationId);
      }
    };
  }, [ref, options]);
};

export default function AppScreens() {
  const [currentIndex, setCurrentIndex] = useState(0);
  const [isAutoPlaying, setIsAutoPlaying] = useState(true);
  const [isMobile, setIsMobile] = useState(false);
  const timeoutRef = useRef<NodeJS.Timeout | null>(null);
  const carouselRef = useRef<HTMLDivElement | null>(null);
  
  // App demo types
  const demos = [
    {
      id: 1,
      type: 'farm',
      title: 'Farm Dashboard',
      description: 'Real-time sales tracking and inventory management'
    },
    {
      id: 2,
      type: 'market',
      title: 'Marketplace',
      description: 'Browse and discover fresh produce from local farms'
    },
    {
      id: 3,
      type: 'delivery',
      title: 'Live Delivery Tracking',
      description: 'Track your orders in real-time from farm to door'
    }
  ];

  // Check for mobile screens
  useEffect(() => {
    const checkForMobile = () => {
      setIsMobile(window.innerWidth < 640);
    };

    checkForMobile();
    window.addEventListener('resize', checkForMobile);
    return () => window.removeEventListener('resize', checkForMobile);
  }, []);

  const resetTimeout = () => {
    if (timeoutRef.current) {
      clearTimeout(timeoutRef.current);
    }
  };

  // Handle auto-play and timer resets
  useEffect(() => {
    resetTimeout();
    
    if (isAutoPlaying) {
      timeoutRef.current = setTimeout(() => {
        setCurrentIndex((prevIndex) => (prevIndex + 1) % demos.length);
      }, 5000);
    }
    
    return () => resetTimeout();
  }, [currentIndex, isAutoPlaying, demos.length]);

  // Add touch swipe functionality
  useEffect(() => {
    const carousel = carouselRef.current;
    if (!carousel) return;

    let touchStartX = 0;
    let touchEndX = 0;
    
    const handleTouchStart = (e: TouchEvent) => {
      touchStartX = e.touches[0].clientX;
    };
    
    const handleTouchMove = (e: TouchEvent) => {
      touchEndX = e.touches[0].clientX;
    };
    
    const handleTouchEnd = () => {
      const swipeDistance = touchEndX - touchStartX;
      if (Math.abs(swipeDistance) > 50) {
        if (swipeDistance > 0) {
          goToPrevious();
        } else {
          goToNext();
        }
      }
    };
    
    carousel.addEventListener('touchstart', handleTouchStart);
    carousel.addEventListener('touchmove', handleTouchMove);
    carousel.addEventListener('touchend', handleTouchEnd);
    
    return () => {
      carousel.removeEventListener('touchstart', handleTouchStart);
      carousel.removeEventListener('touchmove', handleTouchMove);
      carousel.removeEventListener('touchend', handleTouchEnd);
    };
  }, []);

  const goToPrevious = () => {
    setIsAutoPlaying(false);
    const newIndex = currentIndex === 0 ? demos.length - 1 : currentIndex - 1;
    setCurrentIndex(newIndex);
    setTimeout(() => setIsAutoPlaying(true), 8000);
  };

  const goToNext = () => {
    setIsAutoPlaying(false);
    const newIndex = (currentIndex + 1) % demos.length;
    setCurrentIndex(newIndex);
    setTimeout(() => setIsAutoPlaying(true), 8000);
  };

  const goToSlide = (index: number) => {
    setIsAutoPlaying(false);
    setCurrentIndex(index);
    setTimeout(() => setIsAutoPlaying(true), 8000);
  };

  const containerVariants = {
    hidden: { opacity: 0 },
    visible: {
      opacity: 1,
      transition: {
        duration: 0.8,
        staggerChildren: 0.2,
      },
    },
  } as const;

  const itemVariants = {
    hidden: { 
      opacity: 0, 
      y: 30 
    },
    visible: {
      opacity: 1,
      y: 0,
      transition: { 
        duration: 0.7, 
        ease: "easeOut" 
      }
    }
  } as const;

  return (
    <motion.div 
      className="w-full rounded-xl py-8 sm:py-12 px-4 sm:px-6 max-w-6xl mx-auto relative"
      variants={containerVariants}
      initial="hidden"
      animate="visible"
    >
      <motion.h2 
        variants={itemVariants}
        className="text-2xl sm:text-3xl font-bold text-center mb-4"
      >
        Live App Demo
      </motion.h2>
      
      <motion.p 
        variants={itemVariants}
        className="text-gray-600 text-center mb-8 sm:mb-12 max-w-2xl mx-auto"
      >
        Experience our app features with real-time interactive demonstrations
      </motion.p>
      
      <div className="relative" ref={carouselRef}>
        {/* Carousel Container */}
        <div className="flex justify-center items-center">
          {/* Left arrow navigation */}
          <motion.button
            onClick={goToPrevious}
            className="absolute left-0 z-20 transform -translate-y-1/2 top-1/2 bg-white rounded-full p-2 shadow-lg hover:shadow-xl transition-shadow"
            whileHover={{ scale: 1.1 }}
            whileTap={{ scale: 0.9 }}
            aria-label="Previous demo"
            variants={itemVariants}
          >
            <svg width="24" height="24" viewBox="0 0 24 24" fill="none" xmlns="http://www.w3.org/2000/svg" className="w-6 h-6">
              <path d="M15 18l-6-6 6-6" stroke="#666" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"/>
            </svg>
          </motion.button>
          
          {/* Phones Container */}
          <motion.div 
            className="flex justify-center items-center relative w-full overflow-hidden px-16"
            variants={itemVariants}
          >
            <div className="flex gap-8 justify-center items-center">
              {/* Main Center Phone */}
              <motion.div
                className="w-64 h-[512px] relative z-10"
                key={currentIndex}
                initial={{ scale: 0.8, opacity: 0, rotateY: 90 }}
                animate={{ scale: 1, opacity: 1, rotateY: 0 }}
                exit={{ scale: 0.8, opacity: 0, rotateY: -90 }}
                transition={{ duration: 0.6, ease: "easeInOut" }}
              >
                <DemoPhone 
                  demoType={demos[currentIndex].type}
                  animationConfig={{
                    speedX: 0.3,
                    speedY: 0.4,
                    rangeX: 4,
                    rangeY: 8,
                    floatRange: 12
                  }}
                />
                
                {/* Glow effect */}
                <div className="absolute inset-0 bg-gradient-to-br from-green-400 to-blue-500 rounded-[3rem] blur-2xl opacity-20 scale-110 -z-10"></div>
              </motion.div>

              {/* Side Phones (Desktop only) */}
              {!isMobile && (
                <>
                  {/* Left Side Phone */}
                  <motion.div
                    className="w-48 h-96 relative z-5 opacity-60"
                    initial={{ x: -50, opacity: 0 }}
                    animate={{ x: 0, opacity: 0.6 }}
                    transition={{ duration: 0.4, delay: 0.2 }}
                  >
                    <DemoPhone 
                      demoType={demos[(currentIndex - 1 + demos.length) % demos.length].type}
                      className="transform scale-75 -rotate-12"
                      animationConfig={{
                        speedX: 0.2,
                        speedY: 0.3,
                        rangeX: 2,
                        rangeY: 4,
                        floatRange: 6
                      }}
                    />
                  </motion.div>

                  <motion.div

                    initial={{ x: 50, opacity: 0 }}
                    animate={{ x: 0, opacity: 0.6 }}
                    transition={{ duration: 0.4, delay: 0.2 }}
                  >
                    <DemoPhone

              <path d="M9 18l6-6-6-6" stroke="#666" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"/>
